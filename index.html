AFRAME.registerComponent('info-panel', { init: function () { var buttonEls = document.querySelectorAll('.menu-button'); var fadeBackgroundEl = document.querySelector('#fadeBackground'); // Store references to the game images this.gameImageEls = { blasphemousButton: document.querySelector('#blasphemousGameImage'), nineSolsButton: document.querySelector('#nineSolsGameImage'), hollowKnightButton: document.querySelector('#hollowKnightGameImage') }; this.gameTitleEl = document.querySelector('#gameTitle'); this.gameDescriptionEl = document.querySelector('#gameDescription'); // Content mapping for the selected games this.gameInfo = { blasphemousButton: { title: 'Blasphemous (2019)', description: 'A punishing action-platformer that combines the fast-paced, skilled combat of a hack-n-slash game with a deep and evocative narrative core, delivered through exploration of a huge universe comprised of non-linear levels.' }, nineSolsButton: { title: 'Nine Sols (2024)', description: 'A lore-rich, hand-drawn 2D action-platformer featuring Taopunk setting and deflection-focused combat. Embark on a journey of Eastern fantasy, explore the land once ruled by an ancient alien race, and follow a vengeful hero to slay the 9 Sols.' }, hollowKnightButton: { title: 'Hollow Knight (2017)', description: 'Forge your own path in Hollow Knight! An epic action adventure through a vast ruined kingdom of insects and heroes. Explore twisting caverns, battle tainted creatures and befriend bizarre bugs, all in a classic, hand-drawn 2D style.' } }; this.onMenuButtonClick = this.onMenuButtonClick.bind(this); this.onBackgroundClick = this.onBackgroundClick.bind(this); // Add click listeners to all poster menu buttons for (var i = 0; i < buttonEls.length; i++) { buttonEls[i].addEventListener('click', this.onMenuButtonClick); } // Add listener to dismiss the info panel when clicking off this.el.sceneEl.addEventListener('click', this.onBackgroundClick); }, onMenuButtonClick: function (evt) { var buttonEl = evt.currentTarget; var buttonId = buttonEl.getAttribute('id'); var info = this.gameInfo[buttonId]; if (!info) return; // Set title and description text this.gameTitleEl.setAttribute('text', 'value', info.title); this.gameDescriptionEl.setAttribute('text', 'value', info.description); // Hide all image previews first for (var key in this.gameImageEls) { if (this.gameImageEls[key]) { this.gameImageEls[key].setAttribute('visible', false); } } // Show selected game's image if (this.gameImageEls[buttonId]) { this.gameImageEls[buttonId].setAttribute('visible', true); } // Display the info panel with scale animation this.el.setAttribute('visible', true); this.el.setAttribute('animation', { property: 'scale', to: '1 1 1', dur: 200, easing: 'easeOutQuad' }); // Dim the background var fadeBackgroundEl = document.querySelector('#fadeBackground'); if (fadeBackgroundEl) { fadeBackgroundEl.setAttribute('visible', true); } }, onBackgroundClick: function (evt) { // Hide panel if user clicks outside of a menu button if (evt.target.classList.contains('menu-button')) return; this.el.setAttribute('animation', { property: 'scale', to: '0.001 0.001 0.001', dur: 200, easing: 'easeInQuad' }); var fadeBackgroundEl = document.querySelector('#fadeBackground'); if (fadeBackgroundEl) { fadeBackgroundEl.setAttribute('visible', false); } } });
